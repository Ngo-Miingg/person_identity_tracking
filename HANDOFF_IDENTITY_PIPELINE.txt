class TrackInstanceManager:
    """Maps raw tracker IDs to lifecycle-bound immutable instance IDs (Oxxx).

    If a raw tracker ID disappears for longer than max_gap frames and returns,
    it is assigned a brand-new instance ID and starts completely UNBOUND.
    """

    def __init__(self, max_gap: int = 60) -> None:
        self.max_gap = int(max_gap)
        self._next_instance: int = 1
        self._active_instances: dict[tuple[str, int], int] = {}
        self._last_seen: dict[tuple[str, int], int] = {}
        self._closed_instances: set[int] = set()

    def get_or_create(self, raw_tid: int, frame: int, camera: str = "default") -> tuple[int, bool]:
        """Returns (instance_id, is_new)."""
        raw_key = (str(camera), int(raw_tid))
        last = self._last_seen.get(raw_key)
        is_new = False
        if last is None or (int(frame) - last) > self.max_gap or raw_key not in self._active_instances:
            old_instance = self._active_instances.get(raw_key)
            if old_instance is not None:
                self._closed_instances.add(old_instance)
            instance_id = self._next_instance
            self._next_instance += 1
            self._active_instances[raw_key] = instance_id
            is_new = True
        self._last_seen[raw_key] = int(frame)
        return self._active_instances[raw_key], is_new

    def get(self, raw_tid: int, camera: str = "default") -> int | None:
        return self._active_instances.get((str(camera), int(raw_tid)))

    def object_label(self, raw_tid: int, camera: str = "default") -> str:
        inst = self.get(raw_tid, camera)
        return f"O{inst:03d}" if inst is not None else "O:---"

    def mark_ended(self, raw_tid: int, camera: str = "default") -> int | None:
        raw_key = (str(camera), int(raw_tid))
        self._last_seen.pop(raw_key, None)
        instance_id = self._active_instances.pop(raw_key, None)
        if instance_id is not None:
            self._closed_instances.add(instance_id)
        return instance_id

    def is_closed(self, instance_id: int) -> bool:
        return int(instance_id) in self._closed_instances

def safe_crop(frame: np.ndarray, xyxy: np.ndarray) -> np.ndarray | None:
    h, w = frame.shape[:2]
    x1, y1, x2, y2 = [int(round(v)) for v in xyxy]
    x1, x2 = max(0, x1), min(w, x2)
    y1, y2 = max(0, y1), min(h, y2)
    if x2 - x1 < 12 or y2 - y1 < 24:
        return None
    return frame[y1:y2, x1:x2].copy()

def _body_quality(box, frame_shape, det_conf):
    h, w = frame_shape[:2]
    x1, y1, x2, y2 = [float(v) for v in box]
    bw, bh = max(x2 - x1, 0.0), max(y2 - y1, 0.0)
    size = float(np.clip(min(bh / max(h * 0.35, 1.0), bw / max(w * 0.10, 1.0)), 0.0, 1.0))
    clipped = float(x1 <= 1 or y1 <= 1 or x2 >= w - 2 or y2 >= h - 2)
    return float(np.clip(0.60 * float(det_conf) + 0.40 * size - 0.10 * clipped, 0.0, 1.0))

class TorchReIDEmbedder:
    def __init__(self, device="cuda:0", model_name="osnet_x1_0", model_path=None):
        from torchreid.utils import FeatureExtractor
        if device.startswith("cuda") and not torch.cuda.is_available():
            raise RuntimeError("CUDA was requested but torch.cuda.is_available() is False")
        if not model_path:
            raise FileNotFoundError("OSNet checkpoint path is required.")
        model_path = Path(model_path)
        if not model_path.is_file():
            raise FileNotFoundError(f"OSNet checkpoint not found: {model_path}")
        self.extractor = FeatureExtractor(
            model_name=model_name, model_path=str(model_path),
            device=device, verbose=False,
        )

    @torch.inference_mode()
    def encode(self, bgr_images):
        images = []
        for im in bgr_images:
            if im is None or im.size == 0:
                continue
            if im.ndim != 3 or im.shape[2] != 3:
                raise ValueError(f"Expected HxWx3 BGR image, got shape={im.shape}")
            rgb = cv2.cvtColor(im, cv2.COLOR_BGR2RGB)
            images.append(rgb)
        if not images:
            return np.empty((0, 512), dtype=np.float32)
        features = self.extractor(images)
        arr = features if isinstance(features, np.ndarray) else features.detach().float().cpu().numpy()
        arr = arr.astype(np.float32, copy=False)
        arr /= np.clip(np.linalg.norm(arr, axis=1, keepdims=True), 1e-12, None)
        return arr

TorchReID's FeatureExtractor default transform is `image_size=(256,128)`.

def select_body_prototypes(features, max_prototypes=4):
    if features.size == 0:
        return []
    feats = np.asarray(features, dtype=np.float32)
    feats /= np.maximum(np.linalg.norm(feats, axis=1, keepdims=True), 1e-12)
    selected = []
    centroid = normalize(feats.mean(axis=0))
    selected.append(centroid.astype(np.float32))
    while len(selected) < max_prototypes and len(selected) < len(feats):
        sims = np.stack([feats @ p for p in selected], axis=1)
        distance_to_set = 1.0 - sims.max(axis=1)
        idx = int(np.argmax(distance_to_set))
        candidate = feats[idx]
        if max(float(candidate @ p) for p in selected) > 0.985:
            break
        selected.append(candidate.astype(np.float32))
    return selected

File: src/infer_engine.py:54-146

assign_faces_one_to_one(person_boxes, person_keys, faces, face_priors) creates a
cost matrix initialized to 1e9. A face/person candidate is eligible only when:
- face center is within horizontal bbox extent plus 8%;
- face center is between -12% and +50% of person height;
- face width is <= 120% of person width.

Eligible cost:

    dx = abs(face_center_x - person_center_x) / person_width
    dy = abs(face_center_y - (person_y1 + .16 * person_height)) / person_height
    cost = .55 * dx + .45 * dy

Temporal prior IoU > .25 subtracts .30 * IoU from cost. Hungarian is then run
with scipy.optimize.linear_sum_assignment. Cost >= .85 is rejected. If a face is
near-equally eligible for multiple persons (cost margin < .05), status is
AMBIGUOUS and no face is returned.

Output status values: ASSIGNED, NO_FACE, AMBIGUOUS.

12. BODY ROOT-CAUSE ANSWERS
---------------------------
B1. SAME after fragmentation is not reliably separable by OSNet/body alone.
Evidence: CHIRLA SAME centroid min .4246 and median-pairwise median .6349;
cam2 same candidates have strong scores but are not ground truth.

B2. Different people are not rejected reliably by body alone.
Evidence: cam2 simultaneous different O001/O003 max .9836 and O083/O086
centroid .9537, comparable to or above same-person candidates.

B3. No universal single body threshold is proven across all evidence.
On clean CHIRLA pairs only, top5_mean .78 gives 0 wrong/65.38% SAME recall and
max .82 gives 0 wrong/61.54% recall. Cam2 contains different-person max .9836,
so those dataset-local thresholds cannot be promoted to production truth.

B4. Current G continuity is mixed body+motion policy, not attributable to one cue.
There are 2 logged conservative_g_reacquire events and runtime lacks per-cue
scores. Face-specific reacquisition is not recorded for these two events.

B5. Bottleneck is multi-factor:
- crop: whole bbox contains head/background/partial/blurred views;
- aggregation: max similarity can be dominated by one matching frame;
- association: body weight is .65 in conservative score;
- tracker: cam2 has 132 O for 132 local tracks and observed fragmentation;
- logging: per-merge attribution is incomplete.
The current evidence does not isolate OSNet weights from crop effects.

13. OPEN RISKS
-------------
- The separate G object is not implemented; P profile ID is used as G label.
- Body appearance can support O->G even though it cannot confirm EMP.
- Full person crops are vulnerable to background/clothing/partial-body similarity.
- Current runtime does not produce a complete O->G attribution record.
- Ver1 source and cam2 ground truth are missing.
- A long O can contain tracker takeover; current body model evaluation excludes
  mixed-GT O, but production provenance still needs boundary handling.
- `src/occlusion_recovery.py` is shadow-only and has its own raw-ID/O mapping;
  it is not the production identity authority.
